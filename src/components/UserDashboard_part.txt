import {

import { performRealOCR, parseExtractedText, convertPdfToCanvas } from '../utils/ocrScanner';

// Auto-resizing textarea component for text input fields to fit data dynamically without clipping
function AutoResizeTextarea({ className, value, onChange, placeholder, required, disabled }) {
  const textareaRef = useRef(null);

  React.useEffect(() => {
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
      textareaRef.current.style.height = `${textareaRef.current.scrollHeight}px`;
    }
  }, [value]);

  return (
    <textarea
      ref={textareaRef}
      className={className}
      value={value}
      onChange={onChange}
      placeholder={placeholder}
      required={required}
      disabled={disabled}
      rows={1}
      style={{
        resize: 'none',
        overflow: 'hidden',
        minHeight: '38px',
        maxHeight: '200px',
        display: 'block',
        lineHeight: '1.4'
      }}
    />
  );
}
function VerifiedField({ value, rawOcrText, children }) {
  const verify = () => {
    if (!value || !rawOcrText) return null;
    const cleanVal = String(value).replace(/[^a-zA-Z0-9]/g, '').toLowerCase();
    if (cleanVal.length < 3) return null;
    const cleanRaw = rawOcrText.replace(/[^a-zA-Z0-9]/g, '').toLowerCase();
    return cleanRaw.includes(cleanVal);
  };

  const status = verify();

  return (
    <div style={{ position: 'relative', width: '100%', display: 'flex', alignItems: 'center' }}>
      {children}
      {status === true && (
        <CheckCircle2 size={16} color="var(--status-valid)" style={{position: 'absolute', right: '10px', pointerEvents: 'none'}} title="Verified against OCR" />
      )}
      {status === false && (
        <AlertTriangle size={16} color="var(--status-expired)" style={{position: 'absolute', right: '10px', pointerEvents: 'none'}} title="Not found in OCR" />
      )}
    </div>
  );
}

export default function UserDashboard({ employee, onUpdateDocuments }) {

    const timestamp = new Date().toLocaleTimeString();

  };

    switch (status) {
      case 'valid': return 'valid';
      case 'expiring': return 'expiring';

      case 'pending': return 'pending';

      setScanProgress(5);
      addLog(`Selected ${files.length} images. Merging into single layout...`);
      
      try {
        const canvases = await Promise.all(files.map(file => {
          return new Promise((resolve) => {
            const img = new Image();
            img.onload = () => {
              const canvas = document.createElement('canvas');
              canvas.width = img.width;
              canvas.height = img.height;
              canvas.getContext('2d').drawImage(img, 0, 0);
              resolve(canvas);

            img.src = URL.createObjectURL(file);
          });
        }));

        const totalHeight = canvases.reduce((sum, c) => sum + c.height, 0);
        const maxWidth = Math.max(...canvases.map(c => c.width));
        const combinedCanvas = document.createElement('canvas');
        combinedCanvas.width = maxWidth;
        combinedCanvas.height = totalHeight;
        const ctx = combinedCanvas.getContext('2d');
        let currentY = 0;
        for (const c of canvases) {
          ctx.drawImage(c, 0, currentY);
          currentY += c.height;
        }
        
        const dataUrl = combinedCanvas.toDataURL('image/png');
        setPreviewUrl(dataUrl);
        setSelectedFile(files[0]); // Keep first file for reference
        
        // Trigger scan using the array of canvases to prevent Tesseract distortion on massive images
        runOCRScan(canvases, false);
      } catch (err) {
        addLog(`Image merge error: ${err.message || err}`);
        setIsScanning(false);
      }

    }

    setSelectedFile(file);

    const isPdf = file.type === 'application/pdf' || file.name.toLowerCase().endsWith('.pdf');

        runOCRScan(file, false);
      } catch (err) {
        addLog(`PDF conversion error: ${err.message || err}`);
        setIsScanning(false);
      }
